// Archived reference only: email recovery is not part of the active application.
import { useEffect, useState } from "react";
import { api } from "./api";

type RecoveryStatus = {
  email: string | null;
  verifiedAt: string | null;
  deliveryConfigured: boolean;
};

export function AccountSecurity() {
  const [status, setStatus] = useState<RecoveryStatus | null>(null);
  const [email, setEmail] = useState(""),
    [password, setPassword] = useState("");
  const [busy, setBusy] = useState(false),
    [notice, setNotice] = useState(""),
    [error, setError] = useState("");
  useEffect(() => {
    api<RecoveryStatus>("/org/recovery/email")
      .then(setStatus)
      .catch((e) => setError(e.message));
  }, []);
  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setBusy(true);
    setError("");
    setNotice("");
    try {
      const result = await api<{ message: string }>("/org/recovery/email", {
        method: "POST",
        body: JSON.stringify({ email, password }),
      });
      setNotice(result.message);
      setPassword("");
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setBusy(false);
    }
  }
  return (
    <>
      <div className="page-heading">
        <div>
          <div className="eyebrow">Your operator account</div>
          <h1>Account security</h1>
          <p>
            Recovery details belong to your account, not to applicant evidence.
          </p>
        </div>
      </div>
      <section className="panel org-security-panel">
        <h2>Verified recovery email</h2>
        <p>
          {status?.email
            ? `Verified address: ${status.email}`
            : "No verified recovery email has been added."}
        </p>
        {status && !status.deliveryConfigured && (
          <p className="org-form-notice">
            Email delivery is not connected. The server needs SMTP settings
            before verification or reset emails can be sent. Do not rely on
            email recovery yet.
          </p>
        )}
        <p className="muted">
          Confirm your current password, then verify the email link. Links
          expire after 20 minutes and work once. A password reset will sign out
          all your sessions.
        </p>
        <form className="org-form" onSubmit={submit}>
          <label>
            Recovery email
            <input
              aria-label="Recovery email"
              type="email"
              required
              maxLength={254}
              autoComplete="email"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
            />
          </label>
          <label>
            Current password
            <input
              aria-label="Current password"
              type="password"
              required
              minLength={12}
              maxLength={128}
              autoComplete="current-password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
            />
          </label>
          {notice && <p role="status">{notice}</p>}
          {error && (
            <p className="inline-error" role="alert">
              {error}
            </p>
          )}
          <button
            className="btn primary"
            disabled={busy || !status?.deliveryConfigured}
          >
            {busy ? "Sending verification…" : "Send verification link"}
          </button>
        </form>
      </section>
    </>
  );
}

export function RecoveryScreen({
  kind,
  token = "",
  onDone,
}: {
  kind: "request" | "reset" | "verify";
  token?: string;
  onDone: () => void;
}) {
  const [email, setEmail] = useState(""),
    [password, setPassword] = useState(""),
    [confirm, setConfirm] = useState("");
  const [configured, setConfigured] = useState<boolean | null>(null),
    [busy, setBusy] = useState(false),
    [done, setDone] = useState(false);
  const [message, setMessage] = useState(""),
    [error, setError] = useState("");
  useEffect(() => {
    // Keep the secret in component memory, not in page history, request logs, or referrers.
    if (token) history.replaceState({}, "", location.pathname);
    api<{ deliveryConfigured: boolean }>("/org/recovery/config")
      .then((r) => setConfigured(r.deliveryConfigured))
      .catch((e) => setError(e.message));
  }, [token]);
  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setError("");
    if (kind === "reset" && password !== confirm) {
      setError("The two passwords do not match.");
      return;
    }
    setBusy(true);
    try {
      const result = await api<{ message: string }>(`/org/recovery/${kind}`, {
        method: "POST",
        body: JSON.stringify(
          kind === "request"
            ? { email }
            : kind === "reset"
              ? { token, password }
              : { token },
        ),
      });
      setMessage(result.message);
      setPassword("");
      setConfirm("");
      setDone(true);
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setBusy(false);
    }
  }
  return (
    <main className="loading-screen">
      <section className="panel org-security-panel">
        <div className="eyebrow">Identity Lens · Account recovery</div>
        <h1>
          {kind === "request"
            ? "Reset your password"
            : kind === "reset"
              ? "Choose a new password"
              : "Verify recovery email"}
        </h1>
        <p>
          {kind === "request"
            ? "Use the email you previously verified in Account security. An unverified address cannot reset an account."
            : "Only continue if you requested this email. The link expires after 20 minutes and can be used once."}
        </p>
        {kind === "request" && configured === false && (
          <p className="org-form-notice">
            Email delivery is not connected on this server yet. Reset emails
            cannot be sent.
          </p>
        )}
        {!done && (
          <form className="org-form" onSubmit={submit}>
            {kind === "request" && (
              <label>
                Verified recovery email
                <input
                  aria-label="Verified recovery email"
                  type="email"
                  required
                  maxLength={254}
                  autoComplete="email"
                  value={email}
                  onChange={(e) => setEmail(e.target.value)}
                />
              </label>
            )}
            {kind === "reset" && (
              <>
                <label>
                  New password
                  <input
                    aria-label="New password"
                    type="password"
                    required
                    minLength={12}
                    maxLength={128}
                    autoComplete="new-password"
                    value={password}
                    onChange={(e) => setPassword(e.target.value)}
                  />
                </label>
                <label>
                  Confirm new password
                  <input
                    aria-label="Confirm new password"
                    type="password"
                    required
                    minLength={12}
                    maxLength={128}
                    autoComplete="new-password"
                    value={confirm}
                    onChange={(e) => setConfirm(e.target.value)}
                  />
                </label>
              </>
            )}
            <button
              className="btn primary"
              disabled={busy || (kind === "request" && !configured)}
            >
              {busy
                ? "Working…"
                : kind === "request"
                  ? "Send reset link"
                  : kind === "reset"
                    ? "Change password"
                    : "Verify my email"}
            </button>
          </form>
        )}
        {message && <p role="status">{message}</p>}
        {error && (
          <p className="inline-error" role="alert">
            {error}
          </p>
        )}
        <button className="btn" disabled={busy} onClick={onDone}>
          Return to sign in
        </button>
      </section>
    </main>
  );
}
